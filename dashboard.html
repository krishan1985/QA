import pytest
from playwright.sync_api import Page, expect

class TestLogin:

    def test_valid_login(self, page: Page):
        # Navigate to login page
        page.goto("https://example.com/login")
        expect(page).to_have_title("Login — Example")

        # Fill credentials
        page.get_by_label("Email").fill("user@test.com")
        page.get_by_label("Password").fill("Secure@123")
        page.get_by_role("button", name="Sign In").click()

        # Assert redirect to dashboard
        expect(page).to_have_url("https://example.com/dashboard")
        expect(page.get_by_test_id("welcome-msg")).to_be_visible()

    def test_invalid_login(self, page: Page):
        page.goto("https://example.com/login")
        page.get_by_label("Email").fill("wrong@test.com")
        page.get_by_label("Password").fill("wrongpass")
        page.get_by_role("button", name="Sign In").click()

        # Assert error message
        error = page.get_by_role("alert")
        expect(error).to_be_visible()
        expect(error).to_contain_text("Invalid credentials")

    def test_sql_injection(self, page: Page):
        # Security: SQL injection attempt
        page.goto("https://example.com/login")
        page.get_by_label("Email").fill("' OR 1=1 --")
        page.get_by_role("button", name="Sign In").click()
        expect(page).not_to_have_url("*dashboard*")
